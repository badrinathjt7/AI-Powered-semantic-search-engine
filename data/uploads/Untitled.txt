    <!-- Slide 2: What is Semantic Search -->
    <div class="slide page-break">
        <h1>What is Semantic Search?</h1>
        
        <div class="framework-box why-box">
            <div class="framework-title">🎯 WHY it matters</div>
            <p>Traditional keyword search fails when users don't use exact words. Semantic search understands <strong>intent and meaning</strong>, making search smarter and more natural.</p>
        </div>
        
        <div class="framework-box what-box">
            <div class="framework-title">📊 WHAT it is</div>
            <p><strong>Semantic Search</strong> = Search that understands the <span class="highlight">meaning</span> and <span class="highlight">context</span> of your query, not just matching keywords.</p>
            <p style="margin-top: 10px;">It understands synonyms, related concepts, and user intent.</p>
        </div>
        
        <div class="framework-box how-box">
            <div class="framework-title">⚡ HOW it works</div>
            <p>Uses <strong>AI and Natural Language Processing (NLP)</strong> to convert text into mathematical vectors that capture meaning. Similar meanings have similar vectors.</p>
        </div>

        <div class="key-insight">
            🔑 Key Difference: Keyword search = "Find these exact words" | Semantic search = "Find this meaning"
        </div>
    </div>

    <!-- Slide 3: Lexical vs Semantic Search -->
    <div class="slide page-break">
        <h1>Lexical Search vs Semantic Search</h1>
        
        <div class="vs-section">
            <div class="vs-card" style="border-color: #ef4444;">
                <h2 style="color: #ef4444;">📝 Lexical Search</h2>
                <h3>How it works:</h3>
                <p>Matches exact words/tokens using inverted indexes (TF-IDF, BM25)</p>
                
                <h3 style="margin-top: 20px;">Characteristics:</h3>
                <ul>
                    <li>Keyword/token matching</li>
                    <li>Fast and efficient</li>
                    <li>Misses synonyms</li>
                    <li>No context understanding</li>
                    <li>Spelling sensitive</li>
                </ul>
                
                <div style="background: #fee2e2; padding: 15px; margin-top: 20px; border-radius: 8px;">
                    <strong>Example:</strong><br>
                    Query: "car repair"<br>
                    ❌ Misses: "automobile fix", "vehicle maintenance"<br>
                    ✅ Finds: Only exact "car repair" matches
                </div>
            </div>
            
            <div class="vs-divider">VS</div>
            
            <div class="vs-card" style="border-color: #10b981;">
                <h2 style="color: #10b981;">🧠 Semantic Search</h2>
                <h3>How it works:</h3>
                <p>Understands meaning using vector embeddings (neural networks)</p>
                
                <h3 style="margin-top: 20px;">Characteristics:</h3>
                <ul>
                    <li>Meaning-based matching</li>
                    <li>Finds synonyms</li>
                    <li>Context-aware</li>
                    <li>Intent recognition</li>
                    <li>Handles typos better</li>
                </ul>
                
                <div style="background: #d1fae5; padding: 15px; margin-top: 20px; border-radius: 8px;">
                    <strong>Example:</strong><br>
                    Query: "car repair"<br>
                    ✅ Finds: "automobile fix", "vehicle maintenance", "auto service"<br>
                    Understands: All mean the same thing
                </div>
            </div>
        </div>
        
        <div class="key-insight">
            🔑 Lexical = Exact word matching | Semantic = Meaning understanding
        </div>
    </div>

    <!-- Slide 3A: Contextual vs Semantic Search -->
    <div class="slide page-break">
        <h1>Contextual Search vs Semantic Search</h1>
        
        <div class="vs-section">
            <div class="vs-card" style="border-color: #f59e0b;">
                <h2 style="color: #f59e0b;">📍 Contextual Search</h2>
                <h3>How it works:</h3>
                <p>Uses user context (location, history, time, device) to personalize results</p>
                
                <h3 style="margin-top: 20px;">Characteristics:</h3>
                <ul>
                    <li>User-specific personalization</li>
                    <li>Location-aware</li>
                    <li>History-based</li>
                    <li>Time-sensitive</li>
                    <li>Behavior tracking</li>
                </ul>
                
                <div style="background: #fef3c7; padding: 15px; margin-top: 20px; border-radius: 8px;">
                    <strong>Example:</strong><br>
                    Query: "best restaurants"<br>
                    📍 User in Mumbai → Shows Mumbai restaurants<br>
                    📍 User in Delhi → Shows Delhi restaurants<br>
                    ⏰ 8 AM → Breakfast places<br>
                    ⏰ 8 PM → Dinner places
                </div>
            </div>
            
            <div class="vs-divider">VS</div>
            
            <div class="vs-card" style="border-color: #8b5cf6;">
                <h2 style="color: #8b5cf6;">🧠 Semantic Search</h2>
                <h3>How it works:</h3>
                <p>Understands query meaning independent of user context</p>
                
                <h3 style="margin-top: 20px;">Characteristics:</h3>
                <ul>
                    <li>Meaning-based</li>
                    <li>Context-independent</li>
                    <li>Conceptual understanding</li>
                    <li>Synonym recognition</li>
                    <li>Intent interpretation</li>
                </ul>
                
                <div style="background: #f5f3ff; padding: 15px; margin-top: 20px; border-radius: 8px;">
                    <strong>Example:</strong><br>
                    Query: "best restaurants"<br>
                    🧠 Understands: "top dining", "great eateries", "excellent food places"<br>
                    Same for all users (unless combined with context)
                </div>
            </div>
        </div>
        
        <div class="key-insight">
            🔑 Contextual = WHO is searching (personalization) | Semantic = WHAT they mean (understanding)
        </div>
        
        <div style="background: #eff6ff; padding: 20px; border-radius: 10px; margin-top: 20px;">
            <h3>💡 Best Approach: Combine Both!</h3>
            <p><strong>Semantic search</strong> understands the query meaning, then <strong>contextual filtering</strong> personalizes results based on user context.</p>
            <p style="margin-top: 10px;"><strong>Example:</strong> "pizza places" → Semantic finds all pizza restaurants + Contextual shows ones near you, open now, matching your price preferences.</p>
        </div>
    </div>

    <!-- Slide 3B: Vector vs Semantic Search -->
    <div class="slide page-break">
        <h1>Vector Search vs Semantic Search</h1>
        
        <div style="background: #fef3c7; padding: 25px; border-radius: 10px; margin-bottom: 25px; border: 3px solid #f59e0b;">
            <h2 style="color: #78350f; margin-top: 0;">⚠️ Important Clarification!</h2>
            <p style="font-size: 1.15em;"><strong>Vector search and semantic search are often the SAME thing!</strong></p>
            <p style="margin-top: 10px;">Vector search is the <strong>technical implementation</strong> of semantic search. Let's understand the nuance:</p>
        </div>
        
        <div class="vs-section">
            <div class="vs-card" style="border-color: #8b5cf6;">
                <h2 style="color: #8b5cf6;">🔢 Vector Search</h2>
                <h3>What it is:</h3>
                <p><strong>The technical mechanism</strong> - searching through numerical vectors using similarity metrics</p>
                
                <h3 style="margin-top: 20px;">Focus:</h3>
                <ul>
                    <li>Mathematical operation</li>
                    <li>Similarity calculation (cosine, dot product)</li>
                    <li>Vector databases</li>
                    <li>Embeddings storage & retrieval</li>
                    <li>Technical implementation</li>
                </ul>
                
                <div style="background: #f5f3ff; padding: 15px; margin-top: 20px; border-radius: 8px; font-size: 0.9em;">
                    <strong>Technical View:</strong>
                    <div class="code-block" style="font-size: 0.8em; margin-top: 10px;">
query_vec = [0.8, 0.2, 0.5, ...]
doc_vec = [0.75, 0.25, 0.48, ...]
similarity = cosine(query_vec, doc_vec)
# Result: 0.95 (95% similar)
                    </div>
                </div>
            </div>
            
            <div class="vs-divider">=</div>
            
            <div class="vs-card" style="border-color: #10b981;">
                <h2 style="color: #10b981;">🧠 Semantic Search</h2>
                <h3>What it is:</h3>
                <p><strong>The user-facing capability</strong> - understanding meaning and finding conceptually similar content</p>
                
                <h3 style="margin-top: 20px;">Focus:</h3>
                <ul>
                    <li>User experience</li>
                    <li>Meaning understanding</li>
                    <li>Natural language queries</li>
                    <li>Intent recognition</li>
                    <li>Business outcome</li>
                </ul>
                
                <div style="background: #d1fae5; padding: 15px; margin-top: 20px; border-radius: 8px; font-size: 0.9em;">
                    <strong>User View:</strong><br><br>
                    User searches: "laptop for coding"<br><br>
                    ✅ Finds: "programming workstation", "developer machine", "coding laptop"<br><br>
                    User doesn't see vectors - just relevant results!
                </div>
            </div>
        </div>
        
        <div class="key-insight">
            🔑 Vector Search = HOW it works (technology) | Semantic Search = WHAT it does (capability)
        </div>
        
        <h2 style="margin-top: 25px;">The Relationship:</h2>
        <div style="background: #eff6ff; padding: 20px; border-radius: 10px; border-left: 5px solid #3b82f6;">
            <p style="font-size: 1.1em; line-height: 1.8;">
                <strong>Semantic search</strong> is typically <strong>implemented using</strong> vector search.<br><br>
                Vector embeddings (from AI models) capture semantic meaning → Vector search finds similar vectors → User experiences semantic search!
            </p>
        </div>
        
        <h2 style="margin-top: 25px;">When They Differ:</h2>
        <table style="margin-top: 15px;">
            <thead>
                <tr>
                    <th>Scenario</th>
                    <th>Description</th>
                </tr>
            </thead>
            <tbody>
                <tr>
                    <td><strong>Pure Vector Search</strong></td>
                    <td>Finding similar images by visual features (not semantic meaning of content)</td>
                </tr>
                <tr>
                    <td><strong>Semantic Without Vectors</strong></td>
                    <td>Knowledge graphs, ontologies (rare, less effective than vectors)</td>
                </tr>
                <tr>
                    <td><strong>Vector for Non-Semantic</strong></td>
                    <td>Recommendation systems using collaborative filtering (similar users, not meanings)</td>
                </tr>
            </tbody>
        </table>
    </div>

    <!-- Slide 5: How Semantic Search Works -->
    <div class="slide page-break">
        <h1>How Semantic Search Works</h1>
        <h2>The Technical Process</h2>
        
        <div class="step-box">
            <span class="step-number">1</span>
            <strong>Text to Embeddings</strong>
            <p>Convert text into numerical vectors (embeddings) using AI models like BERT, GPT, or Sentence Transformers</p>
            <div class="code-block" style="font-size: 0.8em; margin-top: 10px;">
"dog" → AI Model → [0.8, 0.2, 0.5, 0.1, 0.3, ...]
"puppy" → AI Model → [0.75, 0.25, 0.48, 0.12, 0.32, ...]
            </div>
        </div>

        <div class="step-box">
            <span class="step-number">2</span>
            <strong>Store in Vector Database</strong>
            <p>Save all document embeddings in specialized databases (Pinecone, Weaviate, Milvus)</p>
        </div>

        <div class="step-box">
            <span class="step-number">3</span>
            <strong>Query Conversion</strong>
            <p>User's search query is also converted to a vector using the same AI model</p>
        </div>

        <div class="step-box">
            <span class="step-number">4</span>
            <strong>Similarity Calculation</strong>
            <p>Calculate cosine similarity between query vector and all document vectors</p>
            <div class="code-block" style="font-size: 0.8em; margin-top: 10px;">
Similarity Score = cosine_similarity(query_vector, document_vector)
Result: 0.95 (95% similar) → High relevance!
            </div>
        </div>

        <div class="step-box">
            <span class="step-number">5</span>
            <strong>Rank & Return</strong>
            <p>Return documents sorted by similarity score - most relevant first</p>
        </div>
    </div>

        <h1>Demo Code: Building a Simple Search Engine</h1>
        <p><strong>File:</strong> demo_semantic_search_engine.py</p>
        
        <div class="code-block" style="font-size: 0.55em;">
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

class SemanticSearchEngine:
    def __init__(self):
        print("🚀 Initializing Semantic Search Engine...")
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.documents = []
        self.embeddings = None
        
    def add_documents(self, docs):
        """Add documents to search index"""
        self.documents = docs
        print(f"📚 Indexing {len(docs)} documents...")
        self.embeddings = self.model.encode(docs)
        print("✅ Indexing complete!")
        
    def search(self, query, top_k=3):
        """Search for most similar documents"""
        print(f"\n🔍 Searching for: '{query}'")
        
        # Convert query to embedding
        query_embedding = self.model.encode([query])
        
        # Calculate similarities
        similarities = cosine_similarity(query_embedding, self.embeddings)[0]
        
        # Get top k results
        top_indices = np.argsort(similarities)[::-1][:top_k]
        
        results = []
        for idx in top_indices:
            results.append({
                'document': self.documents[idx],
                'score': similarities[idx]
            })
        
        return results

# ============= USAGE EXAMPLE =============
# Create search engine
engine = SemanticSearchEngine()

# Add documents (imagine these are product descriptions)
products = [
    "Wireless Bluetooth headphones with noise cancellation and 30-hour battery life",
    "Gaming laptop with RTX 4060 graphics card and 16GB RAM for developers",
    "Ergonomic office chair with lumbar support and adjustable height",
    "Smartphone with 5G connectivity and 108MP camera for photography",
    "Smart watch with fitness tracking and heart rate monitor",
    "Mechanical keyboard with RGB lighting for gaming and programming",
    "Portable SSD with 1TB storage and USB-C connection",
    "4K webcam for video conferencing and streaming",
]

engine.add_documents(products)

# Test different queries
queries = [
    "headphones for music",
    "laptop for coding",
    "chair for back pain",
    "phone with good camera"
]

for query in queries:
    results = engine.search(query, top_k=2)
    
    print(f"\n📊 Results:")
    for i, result in enumerate(results, 1):
        print(f"\n{i}. Score: {result['score']:.2%}")
        print(f"   Product: {result['document']}")
    print("\n" + "=" * 70)
